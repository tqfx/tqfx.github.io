192A_INTERN void a_plane_eval(a_plane const *ctx, a_real u, a_real v, a_point3 *res)
193{
194 a_point3 const *const o = &ctx->org;
195 a_vector3 const *const a = &ctx->u_;
196 a_vector3 const *const b = &ctx->v_;
197 res->x = o->x + a->x * u + b->x * v;
198 res->y = o->y + a->y * u + b->y * v;
199 res->z = o->z + a->z * u + b->z * v;
200}
